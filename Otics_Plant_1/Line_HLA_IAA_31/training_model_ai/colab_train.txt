from ultralytics import YOLO
# Load model dengan pretrained weights
model = YOLO('yolov8m.pt')

# Training dengan parameter yang dioptimasi
results = model.train(
    data='/content/drive/MyDrive/Otics Indonesia/Tema_Perbaikan_Kamera_HLA_IAA_31/data.yaml',
    epochs=100,
    imgsz=640,
    batch=16,           # Sesuaikan dengan GPU memory (16/32/64)
    workers=8,          # Optimasi data loading
    patience=20,        # Early stopping jika tidak ada improvement
    save=True,
    save_period=10,     # Simpan checkpoint setiap 10 epoch
    val=True,           # Validasi setiap epoch
    plots=True,         # Generate plots untuk analisis
    # Augmentasi untuk dataset besar
    hsv_h=0.015,        # Hue augmentation
    hsv_s=0.7,          # Saturation augmentation
    hsv_v=0.4,          # Value augmentation
    degrees=10.0,       # Rotasi
    translate=0.1,      # Translasi
    scale=0.5,          # Scaling
    shear=0.0,          # Shear
    perspective=0.0,    # Perspective
    flipud=0.0,         # Flip up-down
    fliplr=0.5,         # Flip left-right
    mosaic=1.0,         # Mosaic augmentation
    mixup=0.1,          # MixUp augmentation
    copy_paste=0.1      # Copy-paste augmentation
)
